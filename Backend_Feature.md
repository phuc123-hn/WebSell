function renderSpecs(specsData) {
    const container = document.querySelector('.specs-container');
    container.innerHTML = '';
    
    Object.entries(specsData.specs).forEach(([category, spec]) => {
        const categoryDiv = document.createElement('div');
        categoryDiv.className = 'spec-category';
        categoryDiv.dataset.category = category;
        
        // Header
        const headerDiv = document.createElement('div');
        headerDiv.className = 'spec-header';

                 style="background: ${spec.iconBg}; color: ${spec.iconColor};">
                ${spec.iconSymbol}
            </div>
            <h3>${spec.name}</h3>
            <p>${spec.sub}</p>
        `;
        categoryDiv.appendChild(headerDiv);
        
        // Sections
        const sectionsDiv = document.createElement('div');
        sectionsDiv.className = 'spec-sections';
        
        spec.sections.forEach((section, index) => {
            const sectionDiv = document.createElement('div');
            sectionDiv.className = 'spec-section';
            
            let content = `
                <h4>${section.label}</h4>
                <p>${section.val}</p>
            `;
            
            if (section.bar !== undefined) {
                content += `
                    <div class="spec-bar" 
                         style="background: linear-gradient(to right, #eee 0%, ${section.barColor} ${section.bar}%)">
                        <span>${section.bar}%</span>
                    </div>
                `;
            }
            
            if (section.tags) {
                content += `
                    <div class="spec-tags">
                        ${section.tags.map(tag => `<span class="tag">${tag}</span>`).join('')}
                    </div>
                `;
            }
            
            sectionDiv.innerHTML = content;
            sectionsDiv.appendChild(sectionDiv);
            
            // Stagger animation
            setTimeout(() => {
                sectionDiv.style.opacity = '1';
                sectionDiv.style.animation = `slideIn 0.6s ease-out ${index * 0.1}s both`;
            }, 0);
        });
        
        categoryDiv.appendChild(sectionsDiv);
        container.appendChild(categoryDiv);
    });
}
